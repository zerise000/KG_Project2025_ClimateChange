# Retrieve the object property values for each etype

PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX owl: <http://www.w3.org/2002/07/owl#>

SELECT ?Source_EType ?Object_Property ?Target_EType (COUNT(*) AS ?Link_Count)
WHERE {
  # Scans every named graph to ensure no data is missed
  GRAPH ?g {
    ?s ?Object_Property ?o .
    
    # Get the type of the subject
    ?s rdf:type ?Source_EType .
    
    # Get the type of the object (target)
    ?o rdf:type ?Target_EType .
  }

  # Filter for your specific project namespace
  FILTER (STRSTARTS(STR(?Source_EType), "http://www.semanticweb.org/mattia/ontologies/2026/0/untitled-ontology-5/"))
  FILTER (STRSTARTS(STR(?Object_Property), "http://www.semanticweb.org/mattia/ontologies/2026/0/untitled-ontology-5/"))

  # Exclude system-level types to keep the matrix clean
  FILTER (?Source_EType != owl:Thing)
  FILTER (?Target_EType != owl:Thing)
  FILTER (?Object_Property != rdf:type)
}
GROUP BY ?Source_EType ?Object_Property ?Target_EType
ORDER BY ?Source_EType ?Object_Property
