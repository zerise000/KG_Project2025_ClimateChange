PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX owl: <http://www.w3.org/2002/07/owl#>

SELECT DISTINCT ?etype
WHERE {
  ?instance rdf:type ?etype .
  
  # Filter out generic OWL/RDF system classes to show only domain entities
  FILTER (?etype != owl:Thing)
  FILTER (!STRSTARTS(STR(?etype), "http://www.w3.org/1999/02/22-rdf-syntax-ns#"))
}
ORDER BY ?etype
